Create voice_type/stt.py:

```python
"""Speech-to-Text client — Google Cloud STT V2 Chirp 2 batch recognition.

Design decisions:
- Batch (not streaming): full audio available before we need transcript.
- Regional endpoint us-central1: required for Chirp 2 (global endpoint
  does not support V2 API). Configured via AppConfig.stt_region.
- language_codes plural: V2 API parameter name (NOT language_code singular).
- Persistent client: initialized once in __init__, reused per recognize().
  Avoids ~200-500ms connection setup overhead on every request.
"""
from __future__ import annotations

import asyncio
from typing import TYPE_CHECKING

from google.api_core.client_options import ClientOptions
from google.cloud.speech_v2 import SpeechClient
from google.cloud.speech_v2.types import cloud_speech
from loguru import logger

from voice_type.config import AppConfig


class STTClient:
    """Google Cloud STT V2 client using Chirp 2 batch recognition."""

    def __init__(self, config: AppConfig) -> None:
        self._project_id = config.gcp_project_id
        self._region = config.stt_region
        self._client = SpeechClient(
            client_options=ClientOptions(
                api_endpoint=f"{self._region}-speech.googleapis.com"
            )
        )
        self._recognizer = (
            f"projects/{self._project_id}/locations/{self._region}/recognizers/_"
        )
        self._rec_config = cloud_speech.RecognitionConfig(
            explicit_decoding_config=cloud_speech.ExplicitDecodingConfig(
                encoding=cloud_speech.ExplicitDecodingConfig.AudioEncoding.LINEAR16,
                sample_rate_hertz=16_000,
                audio_channel_count=1,
            ),
            language_codes=["ru-RU", "en-US"],
            model="chirp_2",
        )
        logger.info(f"STTClient ready — endpoint: {self._region}-speech.googleapis.com")

    async def recognize(self, audio_bytes: bytes) -> str:
        """Transcribe LINEAR16 audio bytes via Chirp 2 batch recognize.

        Runs the blocking GCP call in asyncio.to_thread to avoid blocking
        the event loop.

        Args:
            audio_bytes: Raw LINEAR16 bytes at 16kHz mono int16.
                         Obtain from: numpy_array.tobytes()

        Returns:
            Transcribed text, or empty string if no speech detected.
        """
        if not audio_bytes:
            logger.warning("STTClient.recognize() called with empty audio")
            return ""

        logger.info(f"Sending {len(audio_bytes)} bytes to STT ({len(audio_bytes)/32000:.2f}s audio)")

        def _call() -> str:
            request = cloud_speech.RecognizeRequest(
                recognizer=self._recognizer,
                config=self._rec_config,
                content=audio_bytes,
            )
            response = self._client.recognize(request=request)
            if not response.results:
                return ""
            # Take the top alternative from the first result
            alternatives = response.results[0].alternatives
            if not alternatives:
                return ""
            transcript: str = alternatives[0].transcript
            return transcript

        raw_text = await asyncio.to_thread(_call)
        logger.info(f"STT result: {repr(raw_text[:80])}")
        return raw_text
```

Create voice_type/llm.py:

```python
"""LLM client — Gemini Flash text cleanup with mandatory wall-clock timeout.

Design decisions:
- concurrent.futures wall-clock timeout: Gemini 2.5 Flash has a confirmed
  socket stall bug (googleapis/python-genai#1893) where the connection hangs
  indefinitely without raising a network error. asyncio.wait_for() does NOT
  catch this — the socket stays open. Only a ThreadPoolExecutor future with
  Future.result(timeout=N) provides true wall-clock cancellation.
- thinking_budget=0: 2.5 Flash enables thinking by default, adding 1-3s
  latency. Cleanup is a simple rewrite task — no benefit from thinking.
- Fallback: if Gemini times out or raises, return raw STT text unchanged.
  The user gets their words, just without cleanup. Silent degradation is
  better than blocking indefinitely.
"""
from __future__ import annotations

import concurrent.futures
from typing import Final

from google import genai
from google.genai import types
from loguru import logger

_CLEANUP_PROMPT_TEMPLATE: Final[str] = (
    "Clean up this voice transcription. Fix punctuation, capitalization, and grammar. "
    "Remove filler words (um, uh, ну, эм, короче, вот). "
    "Output only the corrected text, nothing else.\n\n{text}"
)
_MODEL: Final[str] = "gemini-2.5-flash"
_TIMEOUT_S: Final[float] = 8.0


class LLMClient:
    """Gemini Flash client for voice transcription cleanup."""

    def __init__(self) -> None:
        # genai.Client reads GOOGLE_APPLICATION_CREDENTIALS from env.
        # Initialized once — reused across calls.
        self._client = genai.Client()
        logger.info(f"LLMClient ready — model: {_MODEL}, timeout: {_TIMEOUT_S}s")

    def clean(self, raw_text: str) -> str:
        """Clean up raw STT transcription text via Gemini Flash.

        Applies punctuation, grammar fixes and filler word removal.
        Falls back to raw_text if Gemini times out or raises an exception.

        Args:
            raw_text: Raw transcription string from STTClient.

        Returns:
            Cleaned text, or raw_text unchanged on timeout/error.
        """
        if not raw_text.strip():
            return raw_text

        prompt = _CLEANUP_PROMPT_TEMPLATE.format(text=raw_text)

        def _sync_call() -> str:
            response = self._client.models.generate_content(
                model=_MODEL,
                contents=prompt,
                config=types.GenerateContentConfig(
                    thinking_config=types.ThinkingConfig(thinking_budget=0),
                    temperature=0.0,
                ),
            )
            result = response.text
            if result is None:
                return raw_text
            return result.strip()

        with concurrent.futures.ThreadPoolExecutor(max_workers=1) as executor:
            future = executor.submit(_sync_call)
            try:
                cleaned = future.result(timeout=_TIMEOUT_S)
                logger.info(f"LLM cleaned: {repr(raw_text[:40])} -> {repr(cleaned[:40])}")
                return cleaned
            except concurrent.futures.TimeoutError:
                logger.warning(
                    f"LLMClient.clean() timed out after {_TIMEOUT_S}s — "
                    "using raw STT text as fallback"
                )
                return raw_text
            except Exception as exc:
                logger.warning(f"LLMClient.clean() error: {exc} — using raw STT text")
                return raw_text
```
