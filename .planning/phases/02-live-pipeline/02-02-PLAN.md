Create voice_type/paste.py:

```python
"""Paste service — delivers transcribed text into the originally active window.

Design decisions:
- pywin32 (NOT pyautogui): pyautogui adds 200ms+ delay and fails silently
  with Unicode/Russian text in some apps. pywin32 is lower-latency and
  handles CF_UNICODETEXT natively.
- SetForegroundWindow + 50ms sleep: the Win32 focus change is asynchronous —
  keystrokes sent immediately after SetForegroundWindow go to the wrong window.
  50ms is the minimum reliable wait confirmed in Superwhisper production reports.
- Clipboard save/restore: preserves user's clipboard content (PST-02 / Pitfall 6).
  GetClipboardData raises TypeError if clipboard has non-text content (images,
  files) — catch it and skip restoration for those cases.
- hwnd=0: indicates window tracking failed or window is gone. Log and skip.
"""
from __future__ import annotations

import time

import win32api  # type: ignore[import-untyped]
import win32clipboard  # type: ignore[import-untyped]
import win32con  # type: ignore[import-untyped]
import win32gui  # type: ignore[import-untyped]
from loguru import logger


class PasteService:
    """Delivers text into a target window via clipboard + Ctrl+V."""

    def paste(self, text: str, hwnd: int) -> None:
        """Paste text into the window identified by hwnd.

        Sequence:
            1. Save clipboard
            2. Restore focus to hwnd via SetForegroundWindow
            3. Write text to clipboard
            4. Send Ctrl+V
            5. Restore original clipboard

        Args:
            text: Text to paste. Must be non-empty.
            hwnd: Window handle captured at recording stop (REC-04).
                  Use 0 to skip paste (window not available).
        """
        if not text:
            logger.warning("PasteService.paste() called with empty text — skipping")
            return

        if hwnd == 0:
            logger.warning(
                "PasteService.paste(): hwnd=0 — no target window captured. "
                f"Text was: {repr(text[:60])}"
            )
            return

        # Step 1: Save existing clipboard content
        original_clip: str | None = _get_clipboard_text()

        try:
            # Step 2: Restore focus to the original window (PST-02)
            win32gui.SetForegroundWindow(hwnd)
            time.sleep(0.05)  # focus propagation — SetForegroundWindow is async

            # Step 3: Write text to clipboard
            _set_clipboard_text(text)

            # Step 4: Send Ctrl+V
            win32api.keybd_event(win32con.VK_CONTROL, 0, 0, 0)
            win32api.keybd_event(ord("V"), 0, 0, 0)
            win32api.keybd_event(ord("V"), 0, win32con.KEYEVENTF_KEYUP, 0)
            win32api.keybd_event(win32con.VK_CONTROL, 0, win32con.KEYEVENTF_KEYUP, 0)
            time.sleep(0.05)  # allow paste to register before restoring clipboard

            logger.info(f"Pasted into HWND {hwnd}: {repr(text[:60])}")

        finally:
            # Step 5: Restore original clipboard (always attempt, even on error)
            if original_clip is not None:
                _set_clipboard_text(original_clip)
            # If original was None (non-text content), leave clipboard as-is.
            # The user's image/file reference is already gone — no way to restore.


def _get_clipboard_text() -> str | None:
    """Return current clipboard text, or None if clipboard has non-text content."""
    try:
        win32clipboard.OpenClipboard()
        try:
            return str(win32clipboard.GetClipboardData(win32con.CF_UNICODETEXT))
        except TypeError:
            # No text in clipboard (image, file, or empty)
            return None
    except Exception as exc:
        logger.debug(f"Could not read clipboard: {exc}")
        return None
    finally:
        try:
            win32clipboard.CloseClipboard()
        except Exception:
            pass


def _set_clipboard_text(text: str) -> None:
    """Write text to clipboard as CF_UNICODETEXT."""
    win32clipboard.OpenClipboard()
    try:
        win32clipboard.EmptyClipboard()
        win32clipboard.SetClipboardData(win32con.CF_UNICODETEXT, text)
    finally:
        win32clipboard.CloseClipboard()
```

Create voice_type/tray.py:

```python
"""System tray icon — pystray wrapper with idle/recording/processing states.

Design decisions:
- Pre-generate icons at init: PIL image generation is slow (~5ms each).
  Generating on every state change would add visible lag. Three icons at
  startup, then just swap the reference.
- run_detached(): On Windows, pystray is safe to run in any thread.
  run_detached() starts the icon in its own thread and returns immediately,
  letting main.py own the event loop. Do NOT use run() — it blocks.
- set_state() is thread-safe: pystray icon.icon property assignment is
  documented as thread-safe on Windows. No lock needed.
- Quit menu item: the only menu action in Phase 1. Calls the on_quit callback
  which enqueues 'shutdown' into the controller queue.
"""
from __future__ import annotations

from typing import Callable

import pystray  # type: ignore[import-untyped]
from PIL import Image, ImageDraw
from loguru import logger

from voice_type.controller import State

# Icon size in pixels (system tray uses 16x16 or 32x32 — we generate larger
# and let Windows scale down; avoids blurriness on high-DPI displays)
_ICON_SIZE: int = 64

# State colors
_COLORS: dict[State, str] = {
    State.IDLE: "#22c55e",       # green — ready
    State.RECORDING: "#ef4444",  # red — recording in progress
    State.PROCESSING: "#eab308", # yellow — STT + LLM in progress
}


def _make_icon_image(color: str) -> Image.Image:
    """Create a filled circle on transparent background."""
    img = Image.new("RGBA", (_ICON_SIZE, _ICON_SIZE), (0, 0, 0, 0))
    draw = ImageDraw.Draw(img)
    margin = 4
    draw.ellipse(
        (margin, margin, _ICON_SIZE - margin, _ICON_SIZE - margin),
        fill=color,
    )
    return img


class TrayIcon:
    """System tray icon with three visual states (SYS-01, REC-03).

    Usage (from main.py):
        tray = TrayIcon(on_quit=controller.shutdown)
        tray.run_detached()   # non-blocking
        # ... later, state changes ...
        tray.set_state(State.RECORDING)
    """

    def __init__(self, on_quit: Callable[[], None]) -> None:
        # Pre-generate all state icons at startup
        self._icons: dict[State, Image.Image] = {
            state: _make_icon_image(color)
            for state, color in _COLORS.items()
        }

        menu = pystray.Menu(
            pystray.MenuItem("Quit VoiceType", lambda: on_quit())
        )

        self._icon = pystray.Icon(
            name="VoiceType",
            icon=self._icons[State.IDLE],
            title="VoiceType — Idle",
            menu=menu,
        )
        logger.info("TrayIcon initialized")

    def run_detached(self) -> None:
        """Start the tray icon in its own thread. Non-blocking."""
        self._icon.run_detached()
        logger.info("TrayIcon running (detached)")

    def set_state(self, state: State) -> None:
        """Swap the tray icon to reflect the new pipeline state.

        Thread-safe — can be called from any thread.

        Args:
            state: New application state (IDLE / RECORDING / PROCESSING).
        """
        self._icon.icon = self._icons[state]
        self._icon.title = f"VoiceType — {state.name.capitalize()}"
        logger.debug(f"TrayIcon state: {state.name}")

    def stop(self) -> None:
        """Stop the tray icon. Called during shutdown."""
        self._icon.stop()
        logger.info("TrayIcon stopped")
```
